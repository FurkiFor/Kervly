"use strict";
// Kervly dashboard (vanilla JS, no build step). Every value placed into HTML goes through esc().
(() => {
  const BOOT = JSON.parse(document.getElementById("boot").textContent);
  const T = BOOT.T;
  const LANG = BOOT.lang;
  const $ = (s, r = document) => r.querySelector(s);
  const $$ = (s, r = document) => [...r.querySelectorAll(s)];
  const esc = (s) => String(s ?? "").replace(/[&<>"']/g, (c) => "&#" + c.charCodeAt(0) + ";");
  const tr = (k, v = {}) => String(T[k] ?? k).replace(/\{(\w+)\}/g, (_, n) => (v[n] ?? ""));
  const PF = { youtube: "YouTube", instagram: "Instagram", tiktok: "TikTok" };
  const PF_SHORT = { youtube: "YT", instagram: "IG", tiktok: "TT" };
  const ICON = BOOT.icons;

  const S = { csrf: "", brands: [], brand: "", view: "overview", filters: { status: "", platform: "", range: "upcoming" }, errors: 0, media: {} };

  // ---------- helpers ----------
  function toast(msg, isErr) {
    const t = $("#toast");
    t.textContent = msg;
    t.classList.toggle("err", !!isErr);
    t.classList.add("show");
    clearTimeout(t._h);
    t._h = setTimeout(() => t.classList.remove("show"), 4200);
  }
  async function api(path, opts = {}) {
    const r = await fetch(path, {
      ...opts,
      headers: { "Content-Type": "application/json", "X-CSRF-Token": S.csrf, ...(opts.headers || {}) },
      credentials: "same-origin",
    });
    if (r.status === 401) {
      location.href = "/login";
      throw new Error(T.session_over);
    }
    const j = await r.json().catch(() => ({}));
    if (!r.ok) throw new Error(j.error || "HTTP " + r.status);
    return j;
  }
  const post = (path, body) => api(path, { method: "POST", body: JSON.stringify(body || {}) });
  const cur = () => S.brands.find((b) => b.id === S.brand);
  const tz = () => (cur() ? cur().timezone : BOOT.defaultTz);
  const pfBadge = (p) => '<span class="pf ' + p + '" title="' + esc(PF[p]) + '">' + PF_SHORT[p] + "</span>";
  function statusBadge(s) {
    const cls = { scheduled: "info", publishing: "warn", processing: "warn", published: "ok", error: "err", cancelled: "" }[s] || "";
    return '<span class="badge ' + cls + '"><span class="dot"></span>' + esc(T["s_" + s]) + "</span>";
  }
  function fmtTime(iso) {
    return new Intl.DateTimeFormat(LANG, { timeZone: tz(), hour: "2-digit", minute: "2-digit", hourCycle: "h23" }).format(new Date(iso));
  }
  function dayKey(iso) {
    return new Intl.DateTimeFormat("en-CA", { timeZone: tz(), year: "numeric", month: "2-digit", day: "2-digit" }).format(new Date(iso));
  }
  function dayLabel(key) {
    const today = dayKey(new Date().toISOString());
    const tomorrow = dayKey(new Date(Date.now() + 86400000).toISOString());
    const yesterday = dayKey(new Date(Date.now() - 86400000).toISOString());
    if (key === today) return T.today;
    if (key === tomorrow) return T.tomorrow;
    if (key === yesterday) return T.yesterday;
    const [y, m, d] = key.split("-").map(Number);
    return new Intl.DateTimeFormat(LANG, { weekday: "long", day: "numeric", month: "long", timeZone: "UTC" }).format(new Date(Date.UTC(y, m - 1, d, 12)));
  }
  function ago(iso) {
    if (!iso) return T.never;
    const s = Math.round((Date.now() - Date.parse(iso)) / 1000);
    const rtf = new Intl.RelativeTimeFormat(LANG, { numeric: "auto" });
    if (s < 60) return rtf.format(-s, "second");
    if (s < 3600) return rtf.format(-Math.round(s / 60), "minute");
    if (s < 86400) return rtf.format(-Math.round(s / 3600), "hour");
    return rtf.format(-Math.round(s / 86400), "day");
  }
  function empty(icon, text) {
    return '<div class="empty"><div class="em-ico">' + ICON[icon] + "</div>" + esc(text) + "</div>";
  }
  function copyBox(value) {
    return '<div class="copy"><code>' + esc(value) + '</code><button class="btn ghost sm" data-copy="' + esc(value) + '">' + ICON.copy + esc(T.copy) + "</button></div>";
  }

  // ---------- dialogs ----------
  function openDialog(id) {
    const d = document.getElementById(id);
    if (!d.open) d.showModal();
    return d;
  }
  function confirmBox(text, okLabel, danger) {
    return new Promise((resolve) => {
      const d = $("#dlg-confirm");
      $("#cf-text", d).textContent = text;
      const ok = $("#cf-ok", d);
      ok.textContent = okLabel || T.confirm;
      ok.className = "btn" + (danger ? " danger" : "");
      d.onclose = () => resolve(d.returnValue === "ok");
      d.returnValue = "";
      d.showModal();
    });
  }

  async function loadBrands(select) {
    const r = await api("/api/brands");
    S.brands = r.brands;
    S.hosts = r.hosts;
    let saved = "";
    try { saved = localStorage.getItem("brand") || ""; } catch {}
    const has = (id) => S.brands.some((b) => b.id === id);
    S.brand = (select && has(select) && select) || (has(S.brand) && S.brand) || (has(saved) && saved) || (S.brands[0] ? S.brands[0].id : "");
    try { localStorage.setItem("brand", S.brand); } catch {}
    const opts = S.brands.length
      ? S.brands.map((b) => '<option value="' + esc(b.id) + '"' + (b.id === S.brand ? " selected" : "") + ">" + esc(b.name) + "</option>").join("")
      : '<option value="">' + esc(T.no_brand_opt) + "</option>";
    $$(".brand-select").forEach((s) => (s.innerHTML = opts));
  }

  const VIEWS = ["overview", "posts", "brands", "setup", "activity", "account", "admin"];
  function setActive() {
    $$("[data-nav]").forEach((a) => a.classList.toggle("on", a.dataset.nav === S.view));
    $$(".err-count").forEach((c) => {
      c.textContent = S.errors;
      c.classList.toggle("hidden", !S.errors);
    });
  }
  async function route() {
    const h = location.hash.replace(/^#\/?/, "");
    S.view = VIEWS.includes(h) ? h : "overview";
    setActive();
    const main = $("#view");
    main.setAttribute("aria-busy", "true");
    try {
      if (!S.brands.length && !["brands", "setup", "activity", "account", "admin"].includes(S.view)) {
        main.innerHTML = renderWelcome();
      } else {
        main.innerHTML = await ({ overview: viewOverview, posts: viewPosts, brands: viewBrands, setup: viewSetup, activity: viewActivity, account: viewAccount, admin: viewAdmin }[S.view])();
      }
    } catch (e) {
      main.innerHTML = '<div class="card">' + empty("alert", e.message) + "</div>";
    }
    main.removeAttribute("aria-busy");
    window.scrollTo(0, 0);
  }

    return '<div class="page-head"><div><h1>' + esc(title) + "</h1>" + (sub ? '<div class="sub muted small">' + sub + "</div>" : "") + '</div><div class="right row">' + (actions || "") + "</div></div>";
  }

  function renderWelcome() {
    return head(T.welcome_title, esc(T.welcome_sub)) + '<div class="card">' + empty("brand", T.no_brands) +

  // ---------- overview ----------
  async function viewOverview() {
    const o = await api("/api/overview?brand=" + encodeURIComponent(S.brand));
    S.errors = o.counts.errors;
    setActive();
    const c = o.counts;
    const stats =
      '<div class="grid g4">' +
      stat(T.st_today, c.today) + stat(T.st_upcoming, c.upcoming) + stat(T.st_published, c.published_7d) + stat(T.st_errors, c.errors, c.errors > 0) +

      ? '<span class="badge ok"><span class="dot"></span>' + esc(tr("sched_ok", { t: ago(o.scheduler.last_run) })) + "</span>"
      : '<span class="badge warn"><span class="dot"></span>' + esc(o.scheduler.last_run ? tr("sched_late", { t: ago(o.scheduler.last_run) }) : T.sched_never) + "</span>";
    const up = o.upcoming.length
      ? '<div class="plist">' + o.upcoming.map((p) => postRow(p, true)).join("") + "</div>"

      ? '<div class="stack-sm"><h2>' + esc(T.needs_attention) + '</h2><div class="plist">' + o.errors.map((p) => postRow(p, true)).join("") + "</div></div>"

      '<div class="card"><div class="card-head"><h2>' + esc(T.checklist) + "</h2></div>" + checklist(o) + "</div>" +

  }
  function stat(label, n, err) {
    return '<div class="card stat' + (err ? " err" : "") + '"><div class="lab">' + esc(label) + '</div><div class="num">' + esc(n) + "</div></div>";
  }
  function checklist(o) {
    const k = o.checklist;
    const items = [
      { done: true, text: T.ck_login },
      { done: k.allowlist, text: T.ck_allowlist, link: "#/brands" },
      ...k.platforms.map((p) => ({
        done: p.configured && p.connected,
        text: tr(p.configured ? "ck_connect" : "ck_keys", { p: PF[p.platform] }),

      })),
      { done: k.mcp_connected, text: T.ck_mcp, link: "#/setup" },
    ];
    return items
      .map(
        (i) =>
          '<div class="check' + (i.done ? " done" : "") + '"><div class="ci">' + ICON.check + '</div><div class="ct grow">' + esc(i.text) +
          (!i.done && i.link ? ' <a href="' + i.link + '" class="small">' + esc(T.go) + " →</a>" : "") + "</div></div>"
      )
      .join("");
  }

  // ---------- posts ----------
  function postRow(p, compact) {
    const content =
      '<div class="ttl">' + esc(p.title || p.caption.split("\n")[0] || p.media_url.split("/").pop()) + "</div>" +
      '<div class="meta">' + esc(p.media_url.split("/").pop()) + (compact ? " · " + esc(dayLabel(dayKey(p.publish_at))) : "") + "</div>" +
      (p.error ? '<div class="' + (p.status === "error" ? "perr" : "pnote") + '">' + esc(p.error) + "</div>" : "");
    const acts = [];
    if (p.status === "scheduled") acts.push('<button class="btn ghost sm" data-act="edit" data-id="' + esc(p.id) + '" title="' + esc(T.edit) + '">' + ICON.edit + "</button>");
    acts.push('<button class="btn ghost sm" data-act="dup" data-id="' + esc(p.id) + '" title="' + esc(T.duplicate) + '">' + ICON.copy + "</button>");
    if (p.status === "error") acts.push('<button class="btn ghost sm" data-act="retry" data-id="' + esc(p.id) + '">' + ICON.retry + esc(T.retry) + "</button>");
    if (p.status === "scheduled" || p.status === "error") acts.push('<button class="btn ghost sm" data-act="cancel" data-id="' + esc(p.id) + '" title="' + esc(T.cancel) + '">' + ICON.x + "</button>");
    if (p.result_url && p.result_url.startsWith("https://")) acts.push('<a class="btn ghost sm" href="' + esc(p.result_url) + '" target="_blank" rel="noopener noreferrer">' + ICON.external + esc(T.open) + "</a>");
    S.postCache[p.id] = p;
    return (
      '<div class="pitem"><div class="time">' + esc(fmtTime(p.publish_at)) + '</div><div class="pfc">' + pfBadge(p.platform) +
      '</div><div class="content">' + content + '</div><div class="st">' + statusBadge(p.status) + '</div><div class="actions">' + acts.join("") + "</div></div>"
    );
  }
  async function viewPosts() {
    const f = S.filters;
    const q = new URLSearchParams({ brand: S.brand, limit: "200" });
    if (f.status) q.set("status", f.status);
    if (f.platform) q.set("platform", f.platform);
    const now = Date.now();
    if (f.range === "upcoming") q.set("from", new Date(now - 3600000).toISOString());
    if (f.range === "past") q.set("to", new Date(now).toISOString());
    const { posts } = await api("/api/posts?" + q);
    const list = f.range === "upcoming" ? posts.slice().sort((a, b) => a.publish_at.localeCompare(b.publish_at)) : posts;
    const groups = [];
    for (const p of list) {
      const k = dayKey(p.publish_at);
      let g = groups.find((x) => x.k === k);
      if (!g) groups.push((g = { k, items: [] }));
      g.items.push(p);
    }
    const seg = (key, val, label) => '<button class="' + (f[key] === val ? "on" : "") + '" data-filter="' + key + '" data-val="' + val + '">' + esc(label) + "</button>";
    const filters =
      '<div class="row mb18"><div class="seg">' + seg("range", "upcoming", T.upcoming) + seg("range", "past", T.past) + seg("range", "all", T.all) + "</div>" +
      '<select class="select-inline" data-filter-select="status"><option value="">' + esc(T.all_statuses) + "</option>" +
      ["scheduled", "processing", "published", "error", "cancelled"].map((s) => '<option value="' + s + '"' + (f.status === s ? " selected" : "") + ">" + esc(T["s_" + s]) + "</option>").join("") + "</select>" +
      '<select class="select-inline" data-filter-select="platform"><option value="">' + esc(T.all_platforms) + "</option>" +
      Object.keys(PF).map((p) => '<option value="' + p + '"' + (f.platform === p ? " selected" : "") + ">" + PF[p] + "</option>").join("") + "</select></div>";
    const body = groups.length
      ? groups.map((g) => '<div class="day"><div class="day-h"><h3>' + esc(dayLabel(g.k)) + '</h3><span class="tiny faint">' + g.items.length + '</span></div><div class="plist">' + g.items.map((p) => postRow(p)).join("") + "</div></div>").join("")
      : '<div class="plist">' + empty("calendar", T.no_posts) + "</div>";
    return head(T.posts, esc(tr("tz_note", { tz: tz() })), newPostBtn()) + filters + body;
  }

  // ---------- post dialog ----------
  function localParts(iso) {
    const p = Object.fromEntries(
      new Intl.DateTimeFormat("en-CA", { timeZone: tz(), year: "numeric", month: "2-digit", day: "2-digit", hour: "2-digit", minute: "2-digit", hourCycle: "h23" })
        .formatToParts(new Date(iso))
        .map((x) => [x.type, x.value])
    );
    return { date: p.year + "-" + p.month + "-" + p.day, time: p.hour + ":" + p.minute };
  }
  function openPostDialog(mode, p) {
    const d = $("#dlg-post");
    const f = $("#post-form");
    f.reset();
    f.dataset.mode = mode;
    f.dataset.id = p && mode === "edit" ? p.id : "";
    $("#post-title").textContent = mode === "edit" ? T.edit_post : T.new_post;
    $("#post-tz").textContent = tz();
    // Platforms: fixed when editing
    $$(".ptoggle", f).forEach((l) => {
      const inp = $("input", l);
      inp.checked = p ? inp.value === p.platform : inp.value === "youtube";
      inp.disabled = mode === "edit";
      l.classList.toggle("on", inp.checked);
      l.classList.toggle("disabled", mode === "edit" && !inp.checked);
    });
    const when = p && mode === "edit" ? localParts(p.publish_at) : localParts(new Date(Date.now() + 3600000).toISOString());
    f.date.value = when.date;
    f.time.value = mode === "edit" ? when.time : (cur().default_times[0] || when.time);
    f.media_url.value = p ? p.media_url : "";
    f.title.value = p ? p.title : "";
    f.caption.value = p ? p.caption : "";
    f.tags.value = p ? (p.tags || []).join(", ") : "";
    const o = (p && p.options) || {};
    f.yt_privacy.value = o.privacy && p.platform === "youtube" ? o.privacy : "public";
    f.yt_category.value = o.category || "PEOPLE_BLOGS";
    f.yt_kids.checked = !!o.madeForKids;
    f.ig_feed.checked = o.shareToFeed !== false;
    f.tt_comments.checked = !o.disableComment;
    // Default time chips
    $("#time-chips").innerHTML = (cur().default_times || []).map((t) => '<button type="button" class="chip" data-time="' + esc(t) + '">' + esc(t) + "</button>").join("");
    $("#date-chips").innerHTML =
      '<button type="button" class="chip" data-day="0">' + esc(T.today) + '</button><button type="button" class="chip" data-day="1">' + esc(T.tomorrow) + "</button>";

  function syncChips() {
    const f = $("#post-form");
    $$("#time-chips .chip").forEach((c) => c.classList.toggle("on", c.dataset.time === f.time.value));
    const today = localParts(new Date().toISOString()).date;
    const tomorrow = localParts(new Date(Date.now() + 86400000).toISOString()).date;
    $$("#date-chips .chip").forEach((c) => c.classList.toggle("on", (c.dataset.day === "0" ? today : tomorrow) === f.date.value));
  }
  function updateCounters() {
    const f = $("#post-form");
    for (const [name, max] of [["title", 100], ["caption", 2200]]) {
      const el = $("#cnt-" + name);
      const n = f[name].value.length;
      el.textContent = n + " / " + max;
      el.classList.toggle("over", n > max);
    }
  }

    const f = $("#post-form");
    const on = $$('input[name="pf"]', f).filter((i) => i.checked).map((i) => i.value);
    $$("[data-pf-section]", f).forEach((s) => s.classList.toggle("hidden", !on.includes(s.dataset.pfSection)));

  }
  async function loadLibrary() {
    const box = $("#lib");
    const id = S.brand;
    try {
      if (!S.media[id]) S.media[id] = (await api("/api/brands/" + encodeURIComponent(id) + "/media")).media;
      const items = S.media[id];
      if (!items.length) return;
      box.innerHTML = items
        .slice(0, 60)
        .map((m) => '<button type="button" data-media="' + esc(m.url) + '"><span class="mono">' + esc(m.name) + '</span><span class="faint tiny">' + (m.size / 1048576).toFixed(1) + " MB</span></button>")
        .join("");
      box.classList.remove("hidden");
      markLibrary();
    } catch (e) {
      box.classList.add("hidden");
    }
  }
  function markLibrary() {
    const v = $("#post-form").media_url.value;
    $$("#lib button").forEach((b) => b.classList.toggle("on", b.dataset.media === v));
  }
  function uploadFile(file) {
    const name = file.name.replace(/\.[^.]+$/, "").normalize("NFKD").replace(/[^A-Za-z0-9._-]+/g, "-").replace(/^-+|-+$/g, "").slice(0, 100) + ".mp4";
    const status = $("#upload-status");
    status.classList.remove("hidden");
    status.innerHTML = esc(tr("uploading", { name, p: 0 })) + '<div class="progress"><i></i></div>';
    return new Promise((resolve, reject) => {
      const xhr = new XMLHttpRequest();
      xhr.open("POST", "/api/brands/" + encodeURIComponent(S.brand) + "/upload?name=" + encodeURIComponent(name));
      xhr.setRequestHeader("X-CSRF-Token", S.csrf);
      xhr.setRequestHeader("Content-Type", "video/mp4");
      xhr.upload.onprogress = (e) => {
        if (!e.lengthComputable) return;
        const p = Math.round((e.loaded / e.total) * 100);
        status.firstChild.textContent = tr("uploading", { name, p });
        $(".progress i", status).style.width = p + "%";
      };
      xhr.onload = () => {
        let j = {};
        try { j = JSON.parse(xhr.responseText); } catch {}
        if (xhr.status === 200) resolve(j);
        else reject(new Error(j.error || "HTTP " + xhr.status));
      };
      xhr.onerror = () => reject(new Error("Network error"));
      xhr.send(file);
    });
  }
  $("#upload-file").addEventListener("change", async (ev) => {
    const file = ev.target.files[0];
    ev.target.value = "";
    if (!file) return;
    try {
      const r = await uploadFile(file);
      $("#post-form").media_url.value = r.url;
      $("#upload-status").textContent = tr("uploaded", { name: r.name });
      delete S.media[S.brand];
      loadLibrary();
    } catch (e) {
      $("#upload-status").textContent = e.message;
      toast(e.message, true);
    }
  });

      } catch (e) { toast(e.message, true); }

      ev.preventDefault();

        await post("/api/account/2fa/confirm", { code: f.code.value });
        toast(T.twofa_enabled);

      } catch (e) { toast(e.message, true); }
    } else if (f.id === "tfa-off") {
      ev.preventDefault();
      try {
        await post("/api/account/2fa/disable", { password: f.password.value });
        toast(T.twofa_disabled);
        route();
      } catch (e) { toast(e.message, true); }
    }
  });

  async function submitPost(ev) {
    ev.preventDefault();
    const f = $("#post-form");
    const btn = $("#post-save");
    btn.disabled = true;
    const options = {
      youtube: { privacy: f.yt_privacy.value, category: f.yt_category.value, madeForKids: f.yt_kids.checked },
      instagram: { shareToFeed: f.ig_feed.checked },
      tiktok: { disableComment: !f.tt_comments.checked },
    };
    const base = {
      media_url: f.media_url.value.trim(),
      title: f.title.value,
      caption: f.caption.value,
      tags: f.tags.value.split(",").map((s) => s.trim()).filter(Boolean),
      publish_at_local: f.date.value + "T" + f.time.value,
    };
    try {
      if (f.dataset.mode === "edit") {
        const p = S.postCache[f.dataset.id];
        await post("/api/posts/" + f.dataset.id + "/edit", { ...base, options: options[p.platform] });
        toast(T.saved);
      } else {
        const platforms = $$('input[name="pf"]', f).filter((i) => i.checked).map((i) => i.value);
        if (!platforms.length) throw new Error(T.pick_platform);
        const r = await post("/api/posts", { brand: S.brand, platforms, ...base, options });
        toast(tr("scheduled_n", { n: r.created.length }) + (r.errors.length ? " · " + r.errors.map((x) => PF[x.platform] + ": " + x.error).join("; ") : ""), r.errors.length > 0);
      }
      $("#dlg-post").close();
      route();
    } catch (e) {
      toast(e.message, true);
    } finally {
      btn.disabled = false;
    }
  }

  document.addEventListener("click", async (ev) => {
    const el = ev.target.closest("[data-act],[data-copy],[data-filter],[data-time],[data-day],[data-media],[data-close]");
    if (!el) return;
    try {
      if (el.dataset.copy) {
        await navigator.clipboard.writeText(el.dataset.copy);
        toast(T.copied);
        return;
      }
      if (el.dataset.close) return el.closest("dialog").close();
      if (el.dataset.filter) {
        S.filters[el.dataset.filter] = el.dataset.val;
        return route();
      }
      if (el.dataset.time) {
        $("#post-form").time.value = el.dataset.time;
        return syncChips();
      }
      if (el.dataset.day) {
        $("#post-form").date.value = localParts(new Date(Date.now() + Number(el.dataset.day) * 86400000).toISOString()).date;
        return syncChips();
      }
      if (el.dataset.media) {
        $("#post-form").media_url.value = el.dataset.media;
        return markLibrary();
      }
      const id = el.dataset.id;
      switch (el.dataset.act) {
        case "new-post":
          return openPostDialog("new", null);
        case "edit":
          return openPostDialog("edit", S.postCache[id]);
        case "dup":
          return openPostDialog("dup", S.postCache[id]);
        case "cancel":
          if (!(await confirmBox(T.confirm_cancel, T.cancel_post, true))) return;
          await post("/api/posts/" + id + "/cancel");
          toast(T.cancelled);
          return route();
        case "retry":
          await post("/api/posts/" + id + "/retry");
          toast(T.requeued);
          return route();
        case "new-brand":
          return openBrandDialog(null);
        case "storage":
          return openStorageDialog(S.brands.find((b) => b.id === id));
        case "tfa-start":
          return startTfa();
        case "user":
          await post("/api/admin/users/" + id + "/" + el.dataset.op);
          toast(T.done);
          return route();
        case "edit-brand":
          return openBrandDialog(S.brands.find((b) => b.id === id));
        case "disconnect":
          if (!(await confirmBox(tr("confirm_disconnect", { x: cur().name + " / " + PF[el.dataset.pf] }), T.remove, true))) return;
          await post("/api/connections/" + encodeURIComponent(S.brand) + "/" + el.dataset.pf + "/delete");
          toast(T.removed);
          return route();
        case "logout":
          await fetch("/logout", { method: "POST", headers: { "X-CSRF-Token": S.csrf } });
          location.href = "/login";
          return;
      }
    } catch (e) {
      toast(e.message, true);
    }
